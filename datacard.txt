MonthsParam =
VAR MinDate = DATE(2020,1,1)
VAR MaxDate = TODAY()
RETURN
DISTINCT(
  SELECTCOLUMNS(
    ADDCOLUMNS(
      CALENDAR( MinDate, MaxDate ),
      "MonthStart", DATE( YEAR([Date]), MONTH([Date]), 1 )
    ),
    "MonthStart", [MonthStart],
    "Label", FORMAT([MonthStart], "MMM yyyy")
  )
)


Bind MonthsParam[MonthStart] → Parameter p_MonthStart.
